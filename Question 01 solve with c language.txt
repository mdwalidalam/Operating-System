#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

// Global variables to store arguments and results
int *global_nums;
int global_count;

double avg_result;
int min_result;
int max_result;

// Thread function to calculate average
void* calculate_average(void* arg) {
    long long sum = 0;
    for (int i = 0; i < global_count; i++) {
        sum += global_nums[i];
    }
    avg_result = (double)sum / global_count;
    pthread_exit(NULL);
}

// Thread function to calculate minimum value
void* calculate_minimum(void* arg) {
    int mn = global_nums[0];
    for (int i = 1; i < global_count; i++) {
        if (global_nums[i] < mn) {
            mn = global_nums[i];
        }
    }
    min_result = mn;
    pthread_exit(NULL);
}

// Thread function to calculate maximum value
void* calculate_maximum(void* arg) {
    int mx = global_nums[0];
    for (int i = 1; i < global_count; i++) {
        if (global_nums[i] > mx) {
            mx = global_nums[i];
        }
    }
    max_result = mx;
    pthread_exit(NULL);
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Usage: %s <num1> <num2> ... <numN>\n", argv[0]);
        printf("Example: %s 90 81 78 95 79 72 85\n", argv[0]);
        return 1;
    }

    global_count = argc - 1;
    global_nums = (int*)malloc(global_count * sizeof(int));
    for (int i = 0; i < global_count; i++) {
        global_nums[i] = atoi(argv[i + 1]);
    }

    pthread_t thread_avg, thread_min, thread_max;

    // Create worker threads
    pthread_create(&thread_avg, NULL, calculate_average, NULL);
    pthread_create(&thread_min, NULL, calculate_minimum, NULL);
    pthread_create(&thread_max, NULL, calculate_maximum, NULL);

    // Wait for all threads to finish
    pthread_join(thread_avg, NULL);
    pthread_join(thread_min, NULL);
    pthread_join(thread_max, NULL);

    // Print final output exactly matching example format
    printf("Output from Thread\n");
    printf("The average value is %.0f, Minimum value is %d, Maximum value is %d\n", 
           avg_result, min_result, max_result);

    free(global_nums);
    return 0;
}
